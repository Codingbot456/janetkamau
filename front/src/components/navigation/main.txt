// Helper to add screens to navigator
const addScreens = (screens, StackComponent) => {
  return screens.map(screenName => {
    const Component = screenMap[screenName];
    if (!Component) return null;
    
    const WrappedComponent = withScreenLayout(Component, screenName);
    const isSeller = isSellerShop(screenName);
    
    return (
      <StackComponent.Screen
        key={screenName}
        name={screenName}
        component={WrappedComponent}
        options={isSeller ? getSellerShopOptions : undefined}
      />
    );
  }).filter(Boolean);
};

const MainNavigator = () => {
  const { width } = useWindowDimensions();
  const isWeb = Platform.OS === 'web';
  const isWebLargeScreen = isWeb && width >= 768;

  const screenOptions = {
    headerShown: false,
    detachInactiveScreens: false,
    animation: isWeb ? 'none' : 'slide_from_right',
  };

  return (
    <Stack.Navigator screenOptions={screenOptions}>
      {addScreens(allScreens, Stack)}
    </Stack.Navigator>
  );
};
